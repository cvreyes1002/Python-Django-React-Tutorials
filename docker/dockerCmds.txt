# Build an image from a Dockerfile
> docker build -t ubuntu-server:26.04 .

# Create and RUN a docker container
> docker run -it --name [container-name] --user creyes ubuntu-server:26.04 /bin/bash

# Create and RUN a docker container and publish a container's port to the host
> docker run -p 9000:8000 -it --name [container-name] --user creyes ubuntu-server:26.04 /bin/bash

# Stop a container
docker stop <container_id_or_name>

# Start an already existing container
> docker start [container-name]

# Restart a container
> docker restart [container-name]

# Start a terminal on a container already running in the background
> docker exec -it --user creyes [container-name] /bin/bash

# Delete a container
docker rm <container_id_or_name>


# Linux commands inside a container
docker exec -it [container-name] [cmd]
    ex: docker exec -it eams-php-app ls -alh /var/www/html



docker run --name my-mysql-server \
  -e MYSQL_ROOT_PASSWORD=password \
  -p 3306:3306 \
  -d mysql:latest




Docker Compose (docker-compose.yml)
See "eams-php-app" as a sample.

# Check the configuration syntax is valid
docker compose config

# Build the image and start both containers
docker compose up -d --build


docker compose exec mysql mysql -u root -p'password' -e "CREATE DATABASE attendance_system;"

docker compose exec -T mysql mysql -u root -p'password' attendance_system < attendance_system.sql

docker compose exec -T mysql mysql -u root -p'password' attendance_system < sample_data.sql
