# Build an image from a Dockerfile
> docker build -t ubuntu-server:26.04 .

# Create and RUN a docker container
> docker run -it --name [container-name] --user creyes ubuntu-server:26.04 /bin/bash

# Create and RUN a docker container and publish a container's port to the host
> docker run -p 9000:8000 -it --name [container-name] --user creyes ubuntu-server:26.04 /bin/bash

# Stop a container
docker stop <container_id_or_name>

# Start an already existing container
> docker start [container-name]

# Restart a container
> docker restart [container-name]

# Start a terminal on a container already running in the background
> docker exec -it --user creyes [container-name] /bin/bash

# Delete a container
docker rm <container_id_or_name>
